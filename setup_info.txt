1. git clone git@github.com:Matt-Teixeira/pg_manage_v2.git
2. Create and populate .env
3. Build your image (if not already) from Dockerfile
~ docker build -t pg_manage .
4. Create database if not already exists
5. Copy Tables and Schemas -> Run: 1_pgdump_tables_to_local.sh

docker run --rm \
  --add-host=host.docker.internal:host-gateway \
  --env-file .env \
  -v "$PWD":/app -w /app \
  --entrypoint bash \
  pg_manage \
  -lc './scripts/azure_to_local_migration/1_pgdump_tables_to_local.sh'

-- this will copy all Table, Schemas, Triggers, Constraints, Indexes, Views, Sequences, and Extensions via CREATE EXTENSION (must exist on the destination)

6. Copy table data -> Run: 2_pgdump_data_to_local.sh
